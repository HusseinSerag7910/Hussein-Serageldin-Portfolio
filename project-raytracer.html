Color TraceRay(const Ray& r, const BVH& scene, int depth)
{
    if (depth <= 0) return Color(0);

    HitRecord rec;
    if (!scene.Hit(r, 0.001f, INFINITY, rec))
        return SampleSky(r.direction);

    Ray scattered;
    Color attenuation;
    Color emitted = rec.material->Emitted(rec.u, rec.v, rec.p);

    if (!rec.material->Scatter(r, rec, attenuation, scattered))
        return emitted;

    return emitted + attenuation * TraceRay(scattered, scene, depth - 1);
}

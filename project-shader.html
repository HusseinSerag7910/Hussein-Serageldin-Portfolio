// Gerstner wave offset, summed across N wave layers
float3 GerstnerOffset(float2 uv, float time)
{
    float3 offset = 0;
    for (int i = 0; i < _WaveCount; i++)
    {
        float2 dir  = _WaveDir[i];
        float  freq = _WaveFreq[i];
        float  phase = dot(dir, uv) * freq + time * _WaveSpeed[i];

        offset.xz += _Steepness[i] * dir * cos(phase);
        offset.y  += sin(phase) * _WaveAmp[i];
    }
    return offset;
}

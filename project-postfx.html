// Apply a 3D LUT to the graded color, with strength blend
half3 ApplyLUT(half3 color, TEXTURE3D(lut), float strength)
{
    half3 uvw = saturate(color) * (LUT_SIZE - 1.0) / LUT_SIZE
                + 0.5 / LUT_SIZE;
    half3 graded = SAMPLE_TEXTURE3D(lut, sampler_LinearClamp, uvw).rgb;
    return lerp(color, graded, strength);
}
